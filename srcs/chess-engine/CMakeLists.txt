cmake_minimum_required(VERSION 3.16)
project(chess_engine VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.70 REQUIRED)
find_package(nlohmann_json 3.7 REQUIRED)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
option(CHESS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(CHESS_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_executable(chess_engine src/chess.cpp src/protocol.cpp src/server.cpp)
target_include_directories(chess_engine PRIVATE include ${Boost_INCLUDE_DIRS})
target_link_libraries(chess_engine PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
include(CTest)
if(BUILD_TESTING)
    add_executable(chess_tests src/chess.cpp tests/chess_tests.cpp)
    target_include_directories(chess_tests PRIVATE include)
    add_test(NAME chess_rules COMMAND chess_tests)
    add_executable(protocol_tests src/chess.cpp src/protocol.cpp tests/protocol_tests.cpp)
    target_include_directories(protocol_tests PRIVATE include)
    target_link_libraries(protocol_tests PRIVATE nlohmann_json::nlohmann_json)
    add_test(NAME json_protocol COMMAND protocol_tests)
endif()
install(TARGETS chess_engine RUNTIME DESTINATION bin)
